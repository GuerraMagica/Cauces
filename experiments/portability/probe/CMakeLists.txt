cmake_minimum_required(VERSION 3.26)
project(CaucesPortabilitySpike LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT ORT_ROOT)
  message(FATAL_ERROR "Set ORT_ROOT to the local ONNX Runtime 1.31.1 package")
endif()
set(SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../source")
set(GENERATED "${CMAKE_CURRENT_SOURCE_DIR}/generated")
add_library(ort SHARED IMPORTED)
if(WIN32)
  set_target_properties(ort PROPERTIES IMPORTED_LOCATION "${ORT_ROOT}/lib/onnxruntime.dll"
    IMPORTED_IMPLIB "${ORT_ROOT}/lib/onnxruntime.lib")
elseif(APPLE)
  set_target_properties(ort PROPERTIES IMPORTED_LOCATION "${ORT_ROOT}/lib/libonnxruntime.1.31.1.dylib")
else()
  set_target_properties(ort PROPERTIES IMPORTED_LOCATION "${ORT_ROOT}/lib/libonnxruntime.so.1.31.1")
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
endif()
add_library(engine STATIC "${SOURCE}/src/dsp.cpp" "${SOURCE}/src/pipeline.cpp"
  "${GENERATED}/models.cpp" "${GENERATED}/model_assets.cpp" "${GENERATED}/wav.cpp"
  "${SOURCE}/src/api.cpp")
target_include_directories(engine PUBLIC "${SOURCE}/src" PRIVATE "${ORT_ROOT}/include")
if(NOT APPLE)
  target_include_directories(engine PRIVATE compat)
endif()
target_compile_options(engine PRIVATE -Wall -Wextra -Wno-deprecated-declarations -ffp-contract=off)
target_link_libraries(engine PUBLIC ort)
if(WIN32)
  target_compile_definitions(engine PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  # The original DSP obtains int64_t transitively on macOS/Linux, not MinGW.
  # Explicit header injection keeps its source and numerical expressions intact.
  target_compile_options(engine PRIVATE -include cstdint)
  target_link_libraries(engine PRIVATE bcrypt)
  target_link_options(engine INTERFACE -static-libgcc -static-libstdc++)
elseif(NOT APPLE)
  target_link_libraries(engine PRIVATE OpenSSL::Crypto)
endif()
add_executable(pipeline_cli "${SOURCE}/tests/pipeline_cli.cpp")
target_link_libraries(pipeline_cli PRIVATE engine)
add_executable(runtime_probe "${SOURCE}/tests/runtime_probe.cpp")
target_link_libraries(runtime_probe PRIVATE engine)
enable_testing()
foreach(name dsp pipeline)
  add_executable(${name}_tests "${SOURCE}/tests/${name}_tests.cpp")
  target_link_libraries(${name}_tests PRIVATE engine)
  add_test(NAME ${name}_contracts COMMAND ${name}_tests)
endforeach()
add_executable(api_tests "${SOURCE}/tests/api_tests.c")
target_link_libraries(api_tests PRIVATE engine)
set_target_properties(api_tests PROPERTIES LINKER_LANGUAGE CXX)
add_test(NAME api_contracts COMMAND api_tests)
if(NOT WIN32)
  add_executable(wav_tests "${SOURCE}/tests/wav_tests.cpp")
  target_link_libraries(wav_tests PRIVATE engine)
  add_test(NAME wav_contracts COMMAND wav_tests)
endif()
