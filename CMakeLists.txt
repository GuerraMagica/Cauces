cmake_minimum_required(VERSION 3.26)
project(Cauces VERSION 0.4.2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OSX_DEPLOYMENT_TARGET "15.0" CACHE STRING "Minimum macOS")
set(ORT_ROOT "${CMAKE_SOURCE_DIR}/vendor/onnxruntime-osx-arm64-1.31.1" CACHE PATH "ONNX Runtime package")
set(ORT_LIBRARY "${ORT_ROOT}/lib/libonnxruntime.1.31.1.dylib" CACHE FILEPATH "ONNX Runtime dylib")
add_library(cauces_security STATIC src/security.cpp)
target_compile_options(cauces_security PRIVATE -Wall -Wextra -Wno-deprecated-declarations)
target_include_directories(cauces_security PUBLIC src)
add_library(ds_models STATIC src/models.cpp src/model_assets.cpp)
target_include_directories(ds_models PUBLIC src PRIVATE "${ORT_ROOT}/include")
target_link_libraries(ds_models PUBLIC "${ORT_LIBRARY}")
target_compile_options(ds_models PRIVATE -Wall -Wextra -Wno-deprecated-declarations -ffp-contract=off)
add_executable(runtime_probe tests/runtime_probe.cpp)
target_link_libraries(runtime_probe PRIVATE ds_models)
enable_testing()
add_library(ds_dsp STATIC src/dsp.cpp)
target_include_directories(ds_dsp PUBLIC src)
target_compile_options(ds_dsp PRIVATE -Wall -Wextra -ffp-contract=off)
add_executable(dsp_tests tests/dsp_tests.cpp)
target_link_libraries(dsp_tests PRIVATE ds_dsp)
add_test(NAME dsp_contracts COMMAND dsp_tests)
add_library(ds_pipeline STATIC src/pipeline.cpp)
target_include_directories(ds_pipeline PUBLIC src)
target_compile_options(ds_pipeline PRIVATE -Wall -Wextra -ffp-contract=off)
target_link_libraries(ds_pipeline PUBLIC ds_dsp ds_models)
add_executable(pipeline_tests tests/pipeline_tests.cpp)
target_link_libraries(pipeline_tests PRIVATE ds_pipeline)
add_test(NAME pipeline_contracts COMMAND pipeline_tests)
add_library(ds_wav STATIC src/wav.cpp)
target_include_directories(ds_wav PUBLIC src)
target_compile_options(ds_wav PRIVATE -Wall -Wextra -ffp-contract=off)
add_executable(wav_tests tests/wav_tests.cpp)
target_link_libraries(wav_tests PRIVATE ds_wav)
add_test(NAME wav_contracts COMMAND wav_tests)
add_executable(pipeline_cli tests/pipeline_cli.cpp)
target_link_libraries(pipeline_cli PRIVATE ds_pipeline ds_wav)
add_executable(dialogue-worker src/worker.cpp src/scratch.cpp)
target_include_directories(dialogue-worker PRIVATE "${CMAKE_SOURCE_DIR}/vendor/json/include")
target_compile_options(dialogue-worker PRIVATE -Wall -Wextra -Wno-deprecated-declarations)
target_link_libraries(dialogue-worker PRIVATE ds_pipeline ds_wav cauces_security)
add_library(dialogue_separator SHARED src/api.cpp)
target_link_libraries(dialogue_separator PRIVATE ds_pipeline)
target_include_directories(dialogue_separator PUBLIC src)
add_executable(api_tests tests/api_tests.c)
target_link_libraries(api_tests PRIVATE dialogue_separator)
add_test(NAME api_contracts COMMAND api_tests)
add_executable(storage_tests tests/storage_tests.cpp src/scratch.cpp)
target_link_libraries(storage_tests PRIVATE ds_dsp cauces_security)
add_executable(cleanup_tests tests/cleanup_tests.cpp src/scratch.cpp)
target_link_libraries(cleanup_tests PRIVATE ds_dsp cauces_security)
add_test(NAME cleanup_identity COMMAND cleanup_tests)
add_test(NAME storage_bounds COMMAND storage_tests)

add_executable(nonregular_copy_tests tests/nonregular_copy.cpp)
target_link_libraries(nonregular_copy_tests PRIVATE cauces_security ds_wav)
add_test(NAME nonregular_copy COMMAND nonregular_copy_tests)
set_tests_properties(nonregular_copy PROPERTIES TIMEOUT 5)

if(APPLE AND EXISTS "/usr/bin/sandbox-exec")
  add_test(NAME wav_without_dev_fd COMMAND /usr/bin/sandbox-exec -p "(version 1)(allow default)(deny file-read* (subpath \"/dev/fd\"))" $<TARGET_FILE:wav_tests> --deny-dev-fd-expected)
  set_tests_properties(wav_without_dev_fd PROPERTIES TIMEOUT 15)
endif()
